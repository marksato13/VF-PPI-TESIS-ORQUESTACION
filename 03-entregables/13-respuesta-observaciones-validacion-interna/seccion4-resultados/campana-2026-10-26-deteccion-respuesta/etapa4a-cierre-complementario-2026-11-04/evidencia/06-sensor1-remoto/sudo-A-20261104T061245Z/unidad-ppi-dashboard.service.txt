[Service]

ExecStart=/home/m4rk/cyberflow/.venv/bin/python /home/m4rk/cyberflow/scripts/engine/dashboard.py \
    --log-path /home/m4rk/cyberflow/logs/motor_decision.log \
    --manifest-path /home/m4rk/cyberflow/artifacts/preliminar/limpio-2026-10/manifest-if-limpio-2026-10.json \
    --detector-name if_limpio_2026_10 \
    --eve-path /var/log/suricata/eve.json \
    --schema /home/m4rk/cyberflow/configs/features/multilayer-v2.json \
    --schema-extra /home/m4rk/cyberflow/configs/features/multilayer-v3.json \
    --descripciones /home/m4rk/cyberflow/configs/features/descripciones.json \
    --dataset /home/m4rk/cyberflow/artifacts/linea-base/multilayer-v3.csv \
    --escenarios /home/m4rk/cyberflow/configs/escenarios.json \
    --config-toml /home/m4rk/cyberflow/configs/cyberflow.local.toml \
    --services ppi-motor.service,ppi-motor-capture.service,suricata.service \
    --calibrado-en-esta-red \
    --usuarios /etc/cyberflow/usuarios.json \
    --clave-sesion /etc/cyberflow/clave-sesion \
    --tls-cert /etc/cyberflow/panel.crt \
    --tls-key /etc/cyberflow/panel.key \
    --auditoria /home/m4rk/cyberflow/logs/panel_auditoria.jsonl \
    --sesion-minutos 480 \
    --intentos-max 5 \
    --bloqueo-minutos 15 \
    --host 10.10.60.11 --port 8788
Restart=always
RestartSec=5
NoNewPrivileges=true
ProtectSystem=strict

# escribia nada y por eso nunca hizo falta, pero el registro de accesos si.
# Sin esta linea la auditoria fallaria en silencio -y un panel con login cuyo
# registro no se escribe es peor que uno sin login, porque aparenta rastro-.
ReadWritePaths=/home/m4rk/cyberflow/logs

[Install]
